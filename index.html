import "jsr:@supabase/functions-js/edge-runtime.d.ts"; import { createClient } from "jsr:@supabase/supabase-js@2"; const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const secretKeysRaw = Deno.env.get("SUPABASE_SECRET_KEYS"); if (!secretKeysRaw) { throw new Error("SUPABASE_SECRET_KEYS bulunamadı."); } const secretKeys = JSON.parse(secretKeysRaw); const secretKey = Object.values(secretKeys)[0] as string; const supabase = createClient( supabaseUrl, secretKey ); const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function response( data: unknown, status = 200 ) { return new Response( JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json", }, } ); } function cleanName(name: string) { return name .trim() .replace(/\s+/g, " ") .slice(0, 20); } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders, }); } if (req.method !== "POST") { return response( { error: "Sadece POST kabul edilir." }, 405 ); } try { const body = await req.json(); const action = String(body.action || ""); /* * ========================== * LİDERLİK TABLOSU * ========================== */ if (action === "leaderboard") { const { data, error } = await supabase .from("scores") .select( "player_name, score, highest_combo, hits, created_at" ) .order( "score", { ascending: false } ) .limit(50); if (error) { console.error(error); return response( { error: "Skor tablosu alınamadı." }, 500 ); } return response({ scores: data || [] }); } /* * ========================== * İSİM KONTROLÜ * ========================== */ if (action === "check_name") { const name = cleanName( String( body.name || "" ) ); if (name.length < 2) { return response( { available: false, error: "İsim en az 2 karakter olmalı." }, 400 ); } const { data, error } = await supabase .from("scores") .select("id") .ilike( "player_name", name ) .limit(1); if (error) { console.error(error); return response( { error: "İsim kontrol edilemedi." }, 500 ); } return response({ available: !data || data.length === 0 }); } /* * ========================== * SKOR KAYDET * ========================== */ if (action === "submit_score") { const name = cleanName( String( body.name || "" ) ); const hits = Number(body.hits); const highestCombo = Number( body.highestCombo ); if (name.length < 2) { return response( { error: "Geçersiz oyuncu adı." }, 400 ); } if ( !Number.isInteger(hits) || hits < 1 || hits > 200000 ) { return response( { error: "Geçersiz vuruş sayısı." }, 400 ); } if ( !Number.isInteger( highestCombo ) || highestCombo < 1 || highestCombo > hits ) { return response( { error: "Geçersiz combo." }, 400 ); } /* * Aynı isim kontrolü. */ const { data: existing } = await supabase .from("scores") .select("id") .ilike( "player_name", name ) .limit(1); if ( existing && existing.length > 0 ) { return response( { error: "Bu isim daha önce kullanılmış." }, 409 ); } /* * SUNUCU TARAFINDA SKOR HESABI * * Tarayıcıdan score almıyoruz. */ let calculatedScore = 0; for ( let i = 1; i <= hits; i++ ) { let multiplier = 1; if (i >= 30) { multiplier = 5; } else if (i >= 20) { multiplier = 4; } else if (i >= 10) { multiplier = 3; } else if (i >= 5) { multiplier = 2; } calculatedScore += 10 * multiplier; } /* * Veritabanına kaydet. */ const { data, error } = await supabase .from("scores") .insert({ player_name: name, score: calculatedScore, highest_combo: highestCombo, hits: hits }) .select() .single(); if (error) { console.error(error); return response( { error: "Skor kaydedilemedi." }, 500 ); } return response({ success: true, score: data.score, highestCombo: data.highest_combo, hits: data.hits }); } return response( { error: "Geçersiz işlem." }, 400 ); } catch (error) { console.error(error); return response( { error: "Sunucu hatası." }, 500 ); } });
